# Isolated platform target; does not include Aurora/GX or the game.
option(LMNX_LTO "Activar LTO en componentes propios" ON)
option(LMNX_O3 "Usar O3 en vez de O2" OFF)
set(LMNX_PGO "OFF" CACHE STRING "OFF, GENERATE o USE")
set(LMNX_PGO_DIR "${CMAKE_BINARY_DIR}/pgo" CACHE PATH "Profiles from the same compiler and build")
add_library(lmnx_optimizacion INTERFACE)
target_compile_options(lmnx_optimizacion INTERFACE -mcpu=cortex-a57 -mtune=cortex-a57
  -ffp-contract=off -fno-fast-math -fno-omit-frame-pointer -ffunction-sections -fdata-sections)
if(LMNX_O3)
  target_compile_options(lmnx_optimizacion INTERFACE -O3)
else()
  target_compile_options(lmnx_optimizacion INTERFACE -O2)
endif()
if(NOT LMNX_PGO STREQUAL "OFF")
  message(FATAL_ERROR "Game PGO uses tools/build_game_switch.py --pgo GENERATE|USE: validates identity, redirects gcda to SD and disables value profiling. This CMake target builds diagnostics only.")
endif()
add_subdirectory("${CMAKE_SOURCE_DIR}/LMNX-Deko3D" "${CMAKE_BINARY_DIR}/LMNX-Deko3D")
file(GLOB _sdl CONFIGURE_DEPENDS sdl3_shim/*.c)
add_library(lmnx_sdl3 STATIC ${_sdl})
target_include_directories(lmnx_sdl3 PUBLIC sdl3_headers/SDL3 PRIVATE sdl3_shim)
target_compile_definitions(lmnx_sdl3 PRIVATE COS_SWITCH_NRO_DIR="/switch/LM-NX")
target_link_libraries(lmnx_sdl3 PUBLIC nx)
add_library(SDL3::SDL3-static ALIAS lmnx_sdl3)
add_library(lmnx_switch_runtime STATIC runtime.cpp thread_policy.c)
target_include_directories(lmnx_switch_runtime PUBLIC . "${CMAKE_SOURCE_DIR}/src")
target_link_libraries(lmnx_switch_runtime PUBLIC lmnx_sdl3 lmnx_optimizacion nx)
target_link_options(lmnx_switch_runtime INTERFACE "LINKER:--wrap=pthread_create")
if(CMAKE_HOST_WIN32)
  find_program(LMNX_PYTHON NAMES py.exe REQUIRED)
  set(_py_args -3)
else()
  find_program(LMNX_PYTHON NAMES python3 REQUIRED)
endif()
execute_process(COMMAND "${LMNX_PYTHON}" ${_py_args} "${CMAKE_SOURCE_DIR}/tools/prepare_aurora_switch.py"
  OUTPUT_VARIABLE LMNX_AURORA_SOURCE OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _aurora_resultado)
if(NOT _aurora_resultado EQUAL 0)
  message(FATAL_ERROR "Could not prepare the Aurora snapshot")
endif()
# Manifest shared with the direct build; pending units remain visible.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime_sources.json" _runtime_json)
string(JSON _runtime_n LENGTH "${_runtime_json}" compilables)
math(EXPR _runtime_ultimo "${_runtime_n} - 1")
set(_runtime_fuentes)
foreach(_i RANGE ${_runtime_ultimo})
  string(JSON _fuente GET "${_runtime_json}" compilables ${_i})
  list(APPEND _runtime_fuentes "${CMAKE_SOURCE_DIR}/${_fuente}")
endforeach()
add_library(lmnx_xxhash STATIC "${CMAKE_SOURCE_DIR}/LMNX-Deko3D/third_party/xxhash/xxhash.c")
target_include_directories(lmnx_xxhash PUBLIC "${CMAKE_SOURCE_DIR}/LMNX-Deko3D/third_party/xxhash")
add_library(lmnx_aot_runtime_core STATIC ${_runtime_fuentes})
set(_imgui "${CMAKE_SOURCE_DIR}/extern/fuentes/imgui-1.91.9b-docking")
add_library(lmnx_imgui STATIC "${_imgui}/imgui.cpp" "${_imgui}/imgui_draw.cpp"
  "${_imgui}/imgui_tables.cpp" "${_imgui}/imgui_widgets.cpp")
target_include_directories(lmnx_imgui PUBLIC "${_imgui}")
target_link_libraries(lmnx_imgui PRIVATE lmnx_optimizacion)
add_library(lmnx_nod STATIC nod/nod_gcn.cpp)
target_include_directories(lmnx_nod PUBLIC nod)
target_link_libraries(lmnx_nod PRIVATE lmnx_optimizacion)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/aurora_sources.json" _servicios_json)
string(JSON _servicios_n LENGTH "${_servicios_json}" servicios)
math(EXPR _servicios_ultimo "${_servicios_n} - 1")
set(_servicios_fuentes)
foreach(_i RANGE ${_servicios_ultimo})
  string(JSON _fuente GET "${_servicios_json}" servicios ${_i})
  list(APPEND _servicios_fuentes "${LMNX_AURORA_SOURCE}/${_fuente}")
endforeach()
add_library(lmnx_aurora_servicios STATIC ${_servicios_fuentes})
target_include_directories(lmnx_aurora_servicios PRIVATE "${LMNX_AURORA_SOURCE}/include"
  "${LMNX_AURORA_SOURCE}/include/dolphin" "${LMNX_AURORA_SOURCE}/lib"
  "${CMAKE_SOURCE_DIR}/extern/fuentes/fmt-12.1.0/include")
target_include_directories(lmnx_aurora_servicios BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/perfilado"
  "${CMAKE_CURRENT_SOURCE_DIR}/nod" "${CMAKE_SOURCE_DIR}/extern/fuentes/abseil-cpp-20240722.0")
target_compile_definitions(lmnx_aurora_servicios PRIVATE AURORA=1 TARGET_PC=1 FMT_HEADER_ONLY=1)
target_link_libraries(lmnx_aurora_servicios PRIVATE lmnx_sdl3 lmnx_optimizacion)
target_include_directories(lmnx_aot_runtime_core PRIVATE . "${CMAKE_SOURCE_DIR}/src"
  "${LMNX_AURORA_SOURCE}/include" "${LMNX_AURORA_SOURCE}/include/dolphin")
target_link_libraries(lmnx_aot_runtime_core PUBLIC lmnx_switch_runtime lmnx_xxhash lmnx_imgui lmnx_aurora_servicios lmnx_nod)
target_compile_definitions(lmnx_aot_runtime_core PRIVATE TARGET_PC=1 LM_STATIC_MODULE=1)
set(LMNX_GAME_GENERATED "" CACHE PATH "Directory generated by prepare_glme01.py")
if(LMNX_GAME_GENERATED)
  if(NOT EXISTS "${LMNX_GAME_GENERATED}/dol.sha1")
    message(FATAL_ERROR "Missing DOL SHA-1 validation; run tools/prepare_glme01.py")
  endif()
  file(READ "${LMNX_GAME_GENERATED}/dol.sha1" _dol_sha1)
  string(STRIP "${_dol_sha1}" _dol_sha1)
  if(NOT _dol_sha1 STREQUAL "77d49225e7d28b9777916cdbe5809345c570ecbc")
    message(FATAL_ERROR "The AOT module must come from GLME01 revision 0")
  endif()
  file(GLOB _game_chunks CONFIGURE_DEPENDS "${LMNX_GAME_GENERATED}/funcs_*.cpp")
  add_library(lmnx_game_aot STATIC ${_game_chunks} "${LMNX_GAME_GENERATED}/func_table.cpp")
  target_include_directories(lmnx_game_aot PRIVATE "${CMAKE_SOURCE_DIR}/src" "${LMNX_GAME_GENERATED}")
  target_link_libraries(lmnx_game_aot PRIVATE lmnx_optimizacion)
endif()

  if(NOT _ipo)

    set_property(TARGET lmnx_game_aot PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()

  OUTPUT "${CMAKE_BINARY_DIR}/ProjectLM-NX-diagnostico.nro")
