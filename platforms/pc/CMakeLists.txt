cmake_minimum_required(VERSION 3.25)
project(ProjectLM_NX_PC LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
include(${CMAKE_BINARY_DIR}/fuentes.cmake)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory(${LMNX_COPIAS}/SDL ${CMAKE_BINARY_DIR}/SDL EXCLUDE_FROM_ALL)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
if(NOT LMNX_DAWN_REFERENCE)
  add_subdirectory(${DEP_GLSLANG} ${CMAKE_BINARY_DIR}/glslang EXCLUDE_FROM_ALL)
endif()
set(ZLIB_COMPAT ON CACHE BOOL "" FORCE)
set(ZLIB_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${DEP_ZLIB_NG} ${CMAKE_BINARY_DIR}/zlib EXCLUDE_FROM_ALL)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${DEP_ZSTD}/build/cmake ${CMAKE_BINARY_DIR}/zstd EXCLUDE_FROM_ALL)
set(PNG_SOURCES png.c pngerror.c pngget.c pngmem.c pngpread.c pngread.c pngrio.c pngrtran.c pngrutil.c pngset.c pngtrans.c pngwio.c pngwrite.c pngwtran.c pngwutil.c)
list(TRANSFORM PNG_SOURCES PREPEND "${DEP_LIBPNG}/")
configure_file(${DEP_LIBPNG}/scripts/pnglibconf.h.prebuilt ${CMAKE_BINARY_DIR}/pnglibconf.h COPYONLY)
add_library(lmnx_png STATIC ${PNG_SOURCES})
target_include_directories(lmnx_png PUBLIC ${DEP_LIBPNG} ${CMAKE_BINARY_DIR} ${DEP_ZLIB_NG} ${CMAKE_BINARY_DIR}/zlib)
target_link_libraries(lmnx_png PUBLIC zlibstatic)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00 TARGET_PC=1 AURORA=1 AURORA_ENABLE_GX=1 AURORA_CACHE_USE_ZSTD=1 LM_STATIC_MODULE=1 LMNX_PC_TEST=1 FMT_HEADER_ONLY=1 NDEBUG=1)
if(LMNX_PAL)
  add_compile_definitions(LM_SWITCH_PAL_MODULE=1)
endif()
if(NOT LMNX_DAWN_REFERENCE)
  add_compile_definitions(LMNX_PC_VULKAN=1)
endif()
add_compile_definitions(LMNX_PC_DATA="${CMAKE_BINARY_DIR}/datos" LMNX_SHADER_BUILD_ID="pc-vulkan")
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math> $<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>)
include_directories(${LMNX_RAIZ}/src ${LMNX_RAIZ}/platforms/pc ${LMNX_RAIZ}/platforms/switch ${LMNX_RAIZ}/platforms/switch/nod
    ${LMNX_RAIZ}/LMNX-Deko3D/include ${LMNX_RAIZ}/LMNX-Deko3D/third_party/plume ${LMNX_RAIZ}/LMNX-Deko3D/third_party/xxhash
    ${LMNX_RAIZ}/LMNX-Deko3D/third_party/switchwaker_translate ${LMNX_COPIAS}/plume-vulkan ${LMNX_COPIAS}/plume-vulkan/contrib/volk
    ${LMNX_COPIAS}/plume-vulkan/contrib/Vulkan-Headers/include ${LMNX_COPIAS}/plume-vulkan/contrib/VulkanMemoryAllocator/include
    ${LMNX_AURORA}/include ${LMNX_AURORA}/include/dolphin ${LMNX_AURORA}/lib
    ${DEP_ABSEIL_CPP} ${DEP_FMT}/include ${DEP_IMGUI} ${DEP_SQLITE} ${DEP_ZSTD}/lib ${DEP_LIBPNG})
if(LMNX_TRACY)
  add_compile_definitions(TRACY_ENABLE)
  include_directories(${DEP_TRACY}/public)
  add_library(lmnx_tracy STATIC ${DEP_TRACY}/public/TracyClient.cpp)
else()
  include_directories(BEFORE ${LMNX_RAIZ}/platforms/switch/perfilado)
endif()
add_library(lmnx_pc_runtime STATIC ${LMNX_RUNTIME} ${LMNX_AOT} diagnostico_estado.cpp)
target_link_libraries(lmnx_pc_runtime PUBLIC SDL3::SDL3-static)
if(LMNX_DAWN_REFERENCE)
  include(referencia.cmake)
  return()
endif()
add_library(lmnx_pc_aurora STATIC ${LMNX_AURORA_SOURCES} ${LMNX_ABSEIL})
# Only GPU startup consumes this identity. Translator changes must not recompile AOT/HLE or reuse the old
# literal pc-vulkan cache.
set_source_files_properties(${LMNX_AURORA}/lib/nativegpu/gpu.cpp PROPERTIES COMPILE_OPTIONS
  "-ULMNX_SHADER_BUILD_ID;-DLMNX_SHADER_BUILD_ID=\"pc-vulkan-${LMNX_SHADER_ID}\"")
set_source_files_properties(${DEP_ABSEIL_CPP}/absl/time/internal/cctz/src/time_zone_lookup.cc PROPERTIES COMPILE_OPTIONS "-include;winstring.h")
# LMNX_AURORA also contains the snapshot path; the manifest uses another name.
add_executable(ProjectLM-NX-PC main.cpp native_gfx_pc.cpp gpu_pc.cpp shader_pc.cpp metricas.cpp imgui_input_pc.cpp
    ${LMNX_RAIZ}/LMNX-Deko3D/src/aurora_gpu.cpp ${LMNX_COPIAS}/plume-vulkan/plume_vulkan.cpp
    ${LMNX_RAIZ}/platforms/switch/imgui_lmnx.cpp ${DEP_IMGUI}/backends/imgui_impl_sdl3.cpp
    ${DEP_IMGUI}/imgui.cpp ${DEP_IMGUI}/imgui_draw.cpp ${DEP_IMGUI}/imgui_tables.cpp ${DEP_IMGUI}/imgui_widgets.cpp
    ${LMNX_RAIZ}/platforms/switch/nod/nod_gcn.cpp ${LMNX_RAIZ}/LMNX-Deko3D/third_party/xxhash/xxhash.c ${DEP_SQLITE}/sqlite3.c)
target_compile_definitions(ProjectLM-NX-PC PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_THREADSAFE=1)
target_link_libraries(lmnx_pc_aurora PUBLIC SDL3::SDL3-static lmnx_png libzstd_static)
set_source_files_properties(${LMNX_AURORA}/lib/card/DolphinCardPath.cpp PROPERTIES COMPILE_DEFINITIONS "UNICODE;_UNICODE")
target_link_libraries(ProjectLM-NX-PC PRIVATE -Wl,--whole-archive lmnx_pc_runtime -Wl,--no-whole-archive lmnx_pc_aurora SDL3::SDL3-static
    ${CMAKE_BINARY_DIR}/libtint_pc.a lmnx_png libzstd_static glslang glslang-default-resource-limits SPIRV psapi dbghelp ws2_32 winmm bcrypt winhttp shell32 ole32 uuid ntdll dxgi wbemuuid)
target_link_options(ProjectLM-NX-PC PRIVATE -static -static-libgcc -static-libstdc++)
add_executable(lmnx_pc_shader_test EXCLUDE_FROM_ALL ${LMNX_RAIZ}/tools/shaders/check_pc_spirv.cpp shader_pc.cpp)
target_link_libraries(lmnx_pc_shader_test PRIVATE glslang glslang-default-resource-limits SPIRV)
target_link_options(lmnx_pc_shader_test PRIVATE -static -static-libgcc -static-libstdc++)
if(LMNX_TRACY)
  target_link_libraries(ProjectLM-NX-PC PRIVATE lmnx_tracy)
endif()
